```javascript
const section = await figma.getNodeByIdAsync('335:845');
const page = await figma.getNodeByIdAsync('78:2');
await figma.setCurrentPageAsync(page);
for (const theme of section.children) {
    const screens = theme.children.filter(n => n.type === 'FRAME' && /^Q/.test(n.name));
    if (screens.length !== 9) throw new Error(theme.name + ': 기본 화면 수 불일치');
    for (const screen of screens) {
        if (screen.width !== 1440 || screen.height !== 960) throw new Error(screen.name);
        if (!screen.findOne(n => n.name === 'Header')) throw new Error('Missing header');
        const navigation = screen.findOne(n => n.name === 'Navigation');
        if (navigation && navigation.children.length !== 2) {
            throw new Error('Inquiry tabs must have exactly two actions');
        }
        for (const text of screen.findAllWithCriteria({types:['TEXT']})) {
            const segments = text.getStyledTextSegments(['fontName']);
            if (segments.some(s => s.fontName.family !== 'Noto Sans KR')) {
                throw new Error(text.id + ': 글꼴 불일치');
            }
            if (/AI|자동 답변|답변 초안/.test(text.characters)) {
                throw new Error(text.id + ': 제외 기능 표현');
            }
        }
    }
}
return {themes: section.children.length, validation: 'passed'};
```
